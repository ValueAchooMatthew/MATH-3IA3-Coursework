\begin{solution}

		x \cdot (x - 2) = 0 \implies \qquad &\langle \text{By the distributive property} \rangle \\\\
		x = 0 \; \text{or} \; x = 2 \; \qquad &\langle \text{By Theorem 2.1.3.b} \rangle \\
	\end{align*}
\end{solution}

\begin{exercise}[9]
	Let $K \coloneq \{s + t\sqrt{2}: s, t \in \mathbb{Q} \}$. Show that $K$ satisfies the following:
\end{exercise}

\begin{parts}
	\item If $x_1, x_2 \in K$, then $x_1 + x_2 \in K$ and $x_1x_2 \in K$.
\end{parts}

\begin{solution}
	Let $x_1 = s_1 + t_1\sqrt{2}, x_2 = s_2 + t_2\sqrt{2} \in K$. We separate the proof of each claim into two parts into two parts.

	\begin{steps}
		\item Prove $x_1 + x_2 \in K$
	\end{steps}

	\begin{proof}
		We can see that:

		\begin{align*}
			x_1 + x_2 &= s_1 + t_1\sqrt{2} + s_2 + t_2\sqrt{2} \\
			&= (s_1 + s_2) + (t_1 + t_2)\sqrt{2} \\
		\end{align*}

		Thus, since $\mathbb{Q}$ is closed under addition, we know that $(s_1 + s_2) \in \mathbb{Q}$ and $(t_1 + t_2) \in \mathbb{Q}$. Therefore, $((s_1 + s_2) + (t_1 + t_2)\sqrt{2}) \in K$ and thus $(x_1 + x_2) \in K$.
	\end{proof}

	\begin{steps}
		\item Prove $x_1x_2 \in K$
	\end{steps}

	\begin{proof}
		We can see that:

		\begin{align*}
			x_1x_2 &= (s_1 + t_1\sqrt{2})(s_2 + t_2\sqrt{2}) \\
			&= s_1s_2 + s_1t_2\sqrt{2} + s_2t_1\sqrt{2} + t_1t_2\sqrt{2}\sqrt{2} \\
			&= s_1s_2 + (s_1t_2 + s_2t_1)\sqrt{2} + t_1t_2(\sqrt{2})^2 \\
			&= s_1s_2 + (s_1t_2 + s_2t_1)\sqrt{2} + 2t_1t_2 \\
			&= (s_1s_2 + 2t_1t_2) + (s_1t_2 + s_2t_1)\sqrt{2} \\
		\end{align*}

		Thus, since $\mathbb{Q}$ is closed under both addition and multiplication, we know that $s_1s_2 + 2t_1t_2 \in \mathbb{Q}$ and $s_1t_2 + s_2t_1 \in \mathbb{Q}$. Therefore, $(s_1s_2 + st_1t_2) + (s_1t_2 + s_2t_1)\sqrt{2} \in K$ and thus $x_1x_2 \in K$.
	\end{proof}
\end{solution}
	
\begin{parts}
	\item If $x \neq 0$ and $x \in K$, then $\frac{1}{x} \in K$.
\end{parts}

\begin{solution}
	\begin{proof}
		Let $x = s + t\sqrt{2} \in K$, where $s$ and $t$ cannot equal 0 simultaneously. Therefore, we have:

		\begin{align*}
			\frac{1}{x} &= \frac{1}{s + t\sqrt{2}} \\
			&= \frac{1}{s + t\sqrt{2}} \cdot \frac{s - t\sqrt{2}}{s - t\sqrt{2}} \\
			&= \frac{s - t\sqrt{2}}{(s + t\sqrt{2})(s - t\sqrt{2})} \\
			&= \frac{s - t\sqrt{2}}{s^2 + (t\sqrt{2})^2} \\
			&= \frac{s - t\sqrt{2}}{s^2 + 2t^2} \\
			&= \frac{s}{s^2 + 2t^2} - \frac{t}{s^2 + 2t^2}\sqrt{2} \\
		\end{align*}

		Therefore, since $\mathbb{Q}$ is closed under addition, multiplication, and division, we know $\frac{s}{s^2 + 2t^2} \in \mathbb{Q}$ and $\frac{t}{s^2 + 2t^2} \in \mathbb{Q}$. Therefore, $\frac{s}{s^2 + 2t^2} - \frac{t}{s^2 + 2t^2}\sqrt{2} \in \mathbb{Q}$ and thus $\frac{1}{x} \in K$ if $x \neq 0$.
	\end{proof}
\end{solution}

			a \; &< \; b \\

		\end{align*}

\begin{exercise}[6]
	Let $S$ be a nonempty subset of $\mathbb{R}$ that is bounded below. Prove that $\text{inf}(S) = -\text{sup}(\{-s: s \in S\})$.
\end{exercise}

\begin{solution}
	\begin{proof}
		We can appeal to the completeness property of $\mathbb{R}$ to assert that $\text{inf}(S)$ must exist since $S$ is bounded below. Let $m = \text{inf}(S)$. By definition of infiumum, we know that for all $s \in S$, $m \le s$. Therefore, we know that $-m \ge -s$. Therefore, for the set $K = \{-s: s \in S\}$, $-m$ upperbounds $K$. Now we must prove that $-m$ is the supremum of $K$. Since $m$ is the infimum of $S$, we know that $\forall \epsilon > 0$, $\exists \; s \in S$ such that $m + \epsilon > s$. Therefore, we have $-m - \epsilon < -s$. Therefore, since $m$ upperbounds $K$, and for arbitrary $\epsilon > 0$, there always exists a $k \in K$ such that $-m -\epsilon < k$, $-m$ must be the supremum of $K = \{-s: s \in S\}$. Therefore, appealing to our earlier definition of $m$, we have:

		\begin{align*}
			m &= m \\
			m &= -(-m) \\
			\text{inf}(S) &= -\text{sup}(K) \\
			\text{inf}(S) &= -\text{sup}(\{-s: s\in S\})\\
		\end{align*}
	\end{proof}
\end{solution}

\begin{exercise}[10]
	Show that if $A$ and $B$ are bounded subsets of $\mathbb{R}$, then $A \cup B$ is a bounded set. Show that $\text{sup}(A \cup B) = \text{sup}(\{\text{sup}(A), \text{sup}(B)\})$.
\end{exercise}

\begin{solution}
	
	We separate our proof into two parts.

	\begin{steps}
		\item Prove that $A \cup B$ is bounded.
	\end{steps}

	\begin{proof}
		By definition of boundedness, we know that $\exists \; M_1, M_2 \in \mathbb{R}$ such that $|a| \le M_1, |b| \le M_2 \forall a, b \in A, B$. Let $M = max{M_1, M_2}$. Thus, we trivially have $M \ge M_1$ and $M \ge M_2$. We know that $\forall c \in (A \cup B)$, $c \in A \vee c \in B$. If $c \in A$, then $|c| \le M$ since $M \ge M_1$ and if $c \in B$, $|c| \le M$ since $M \ge M_2$. Thus, $A \cup B$ must be bounded for bounded $A, B$.
	\end{proof}

	\begin{steps}
		\item Prove that $\text{sup}(A \cup B) = \text{sup}(\{\text{sup}(A), \text{sup}(B)\})$.
	\end{steps}
	
	\begin{proof}
		
		Appealing to the completeness property of $\mathbb{R}$, since $A$ and $B$ are both bounded, we know that $\text{sup}(A)$, $\text{sup}(B)$ both exist. Let $s_1 = \text{sup}(A)$ and $s_2 = \text{sup}(B)$, and $\epsilon > 0$ be arbitrary. By definition of suprema, we know that $\exists a_0, b_0 \in A, B$:

		\begin{align*}
			s_1 - \epsilon < a_0 \quad \text{and} \quad s_2 - \epsilon < b_0 \\
		\end{align*}

		Now, let $s_3 = \text{sup}(\{s_1, s_2\})$.  Since the set $\{s_1, s_2\}$ is finite, we know that the supremum must be an element of the set and thus either $s_3 = s_1$ or $s_3 = s_2$ and therefore $s_3 \ge s_1$ and $s_3 \ge s_2$. Let $c \in A \cup B$. Since $c \in A \vee c \in B$, we know trivially that $c \le s_3$ meaning $s_3$ upperbounds $A \cup B$. To show that $s_3$ must be the supremum, consider again $\epsilon > 0$. From the properties of $s_1$ and $s_2$ that we discussed earlier, we know that if $s_3 = s_1$, $\exists a_0 \in A$ such that $s_3 - \epsilon < a_0$ and if $s_3 = s_2$, $\exists b_0 \in B$ such that $s_3 - \epsilon < b_0$ for all $\epsilon$. Since $a_0, b_0 \in A\cup B$, there always exists a $c_0 \in A \cup B$ such that:

		\begin{align*}
			s_3 - \epsilon < c_0 \\
		\end{align*}

		Thus, by definition of supremum, we know $s_3 = \text{sup}(A \cup B) = \text{sup}(\{\text{sup}(A), \text{sup}(B)\})$.
	\end{proof}
\end{solution}
