\begin{exercise}[5]

\end{exercise}

\begin{solution}
	\begin{step}
		Prove that $|g(x) - c^2| \le 2a|x - c|$ for all $ x, \, c \in I$.
	\end{step}

	\begin{align*}
		|g(x) - c^2| &= |x^2 - c^2| \\
		&= |\left( x + c \right) \left( x - c \right)| \\
		&\le |\left( a + a \right) \left( x - c \right)| \\
		&= |2a\left( x - c \right)| \\
		&= 2a|x - c| \\
	\end{align*}

	\begin{step}
		Prove that $\lim_{x\to{c}}x^2 = c^2$ for all $c \in I$.
	\end{step}

	Let $\epsilon > 0$ be arbitrary. Selecting $\delta = \frac{\epsilon}{2a}$ we can see that, in conjunction with the inequality derived earlier, that for all $x \in I$ satisfying $0 < |x - c| < \delta$:

	\begin{align*}
		|g(x) - c^2| &\le 2a|x - c| \\
		&< 2a \cdot \frac{\epsilon}{2a} \\
		&= \epsilon \\
	\end{align*}

	Thus, by definition of functional limits, we know that $\lim_{x\to{c}}g(x) = \lim_{x\to{c}} x^2 = c^2$ for arbitrary $c \in I$.

\begin{exercise}[14]
	Let $c \in \mathbb{R}$ and let $f: \mathbb{R} \rightarrow \mathbb{R}$ be such that $\lim_{x\to{c}} (f(x))^2 = L$.
\end{exercise}

\begin{subexercise}
	Show that if $L = 0$, then $\lim_{x\to{c}}f(x) = 0$.
\end{subexercise}

\begin{solution}
	If $L = 0$, then for arbitrary $\epsilon > 0$, there exists a $\delta > 0$ such that all $x$ satisfying $0 < |x - c| < \delta$ also satisfies $|\left( f(x) \right)^2 - L| < \epsilon^2$. However, note that if $L = 0$, then:

	\begin{align*}
		|\left( f(x) \right)^2 - L| &< \epsilon^2 \\
		|\left( f(x) \right)^2 - 0| &< \epsilon^2 \\
		|\left( f(x) \right)^2| &< \epsilon^2 \\
		|f(x)|^2 &< \epsilon^2 \\
		|f(x)| &< \sqrt{\epsilon^2} \\
		|f(x)| &< \epsilon \\
		|f(x) - 0| &< \epsilon \\
	\end{align*}

	Thus, since for all $\epsilon > 0$, there exists a $\delta > 0$ such that all $x$ satisfying $0 < |x - c| < \delta$ also satisfies $|f(x) - 0| < \epsilon$, then by definition of functional limits, $\lim_{x\to{c}}f(x) = L$.
\end{solution}

\end{subexercise}

	If $\left(f(x) \right)^2 = x$, we can trivially see that $\lim_{x\to{1}}\left( f(x) \right)^2 = \lim_{x\to{1}} x = 1$ and $\lim_{x\to{1}} f(x) = \lim_{x\to{1}} \sqrt{x} = 1$. However, if $\left( f(x) \right)^2 = \frac{1}{x - 1}$ then $\lim_{x\to{0}} \left( f(x) \right )^2 = \lim_{x\to{0}} \frac{1}{x - 1} = -1$, however since $f(x) = \frac{1}{\sqrt{1}{x - 1}}$ has an interval around $x = -1$ for which all $x$ values in that interval do not belong to the domain of $f$, $\lim_{x\to{0}} f(x) = \lim_{x\to{0}} \frac{1}{\sqrt{x - 1}}$ does not exist.
\end{solution}

\begin{exercise}[4]
	Prove that $\lim_{x\to{0}} \cos \left( \frac{1}{x} \right)$ does not exist but that $\lim_{x\to{0}}x \cos \left( \frac{1}{x} \right) = 0$.
\end{exercise}

\begin{solution}
	\begin{step}
		Prove that $\lim_{x\to{0}} \cos \left( \frac{1}{x} \right)$ does not exist.
	\end{step}

	\begin{proof}
		Let $(x_n)$ be the sequence $x_n = \frac{1}{2\pi{n}}$ and $(y_n)$ be the sequence $y_n = \frac{1}{(2n - 1)\pi}$. By algebraic limit rules, we know that $\lim_{n\to\infty} (x_n) = \lim_{n\to\infty} (y_n) = 0$. Now consider the limits of sequences $\cos \left( \frac{1}{(x_n)} \right)$ and $\cos \left( \frac{1}{(y_n)} \right)$ as $n$ approaches infinity. We can see that:


		\begin{align*}
			\lim_{n\to\infty}\cos \left( \frac{1}{(x_n)} \right) &= \lim_{n\to\infty} \cos \left( 2\pi{n} \right) \qquad \lim_{n\to\infty} \cos \left( \frac{1}{y_n} \right) &&= \lim_{n\to\infty} \cos \left( (2n - 1)\pi \right) \\
			&= 1 &&= -1 \\
		\end{align*}

		Therefore, since there exists two different sequences both converging to zero for which the function does not converge to the same result, by the sequential definition of functional limits, the limit does not exist.
	\end{proof}

	\begin{step}
		Prove that $\lim_{x\to{0}} x\cos \left( \frac{1}{x} \right) = 0$.
	\end{step}

	\begin{proof}
		Since $-1 \le \cos \left( \frac{1}{x} \right) \le 1$, we can establish that $-x \le x \cos \left( \frac{1}{x} \right) \le x$. Now, since $\lim_{x\to{0}} -x = \lim_{x\to{0}} x = 0$, by the Squeeze Theorem, we know that $\lim_{x\to{0}} x\cos \left( \frac{1}{x} \right) = 0$
	\end{proof}
\end{solution}

\begin{exercise}[5]
	Let $f, \, g$ be defined on $A \subseteq \mathbb{R}$ to $\mathbb{R}$, and let $c$ be a cluster point of $A$. Suppose that $f$ is bounded on a neighbourhood of $c$ and that $\lim_{x\to{c}} g = 0$. Prove that $\lim_{x\to{c}} f\cdot{g} = 0$.
\end{exercise}

\begin{solution}
	\begin{proof}
		Let $\delta_0$ be the neighbourhood around $c$ for which $f$ is bounded. Thus, there exists some $M \in \mathbb{R}$ such that $M > 0$ for which $\forall x \in V_{\delta_0}(c)$, $|f(x)| < M$. Let $\epsilon > 0$ be arbitrary. Since $\lim_{x\to{c}} g = 0$, we know by the definition of functional limits that there exists some $\delta_1$ such that for all $x$ satisfying $0 < |x - c| < \delta_1$, then $g$ satisfies $|g(x) - 0| = |g(x)| < \frac{\epsilon}{M}$. Letting $\delta = \min{\delta_0, \delta_1}$, we therefore know that for all $x$ satisfying $0 < |x - c| < \delta$:

		\begin{align*}
			|f(x)g(x) - 0| &= |f(x)g(x)| \\
			&= |f(x)||g(x)| \\
			&< M \cdot \frac{\epsilon}{M} \\
			&= \epsilon
		\end{align*}

		Thus, since for arbitrary $\epsilon > 0$, there always exists some $\delta > 0$ for which $x$ satisfying $0 < |x - c| < \delta$ implies $|f(x)g(x) - 0| < \epsilon$, then by definition of functional limits, $\lim_{x\to{c}}f(x)g(x) = 0$.
	\end{proof}
\end{solution}

\begin{exercise}[8]
	Let $n \in \mathbb{N}$ be such that $n \ge 3$. Derive the inequality $-x^2 \le x^n \le x^2$ for $-1 < x < 1$. Then use the fact that $\lim_{x\to{0}}x^2 = 0$ to show that $\lim_{x\to{0}}x^n = 0$.
\end{exercise}

\begin{solution}
	\begin{step}
		Derive the inequality $-x^2 \le x^n \le x^2$ for $-1 < x < 1$.
	\end{step}

	Since $n \ge 3$, we know that there exists some $k \in \mathbb{N}$ such that $n - k = 2$. Therefore, $x^n = x^2x^{k}$. Since $-1 < x < 1$, we know that $-1 \le x^k \le 1$ and thus $-x^2 \le x^2x^{k} \le x^2$. Therefore, since $x^n = x^2x^k$, we can conclude that $-x^2 \le x^n \le x^2$.

	\begin{step}
		Show that $\lim_{x\to{0}} x^n = 0$.
	\end{step}

	Using our prior inequality, we know that since $-x^2 \le x^n \le x^2$, thus, we know that $\lim_{x\to{0}}-x^2 \le x^n \le \lim_{x\to{0}} x^2$. Since we know that $\lim_{x\to{0}} x^2 = 0$, by algebraic limit theorems, we can conclude that $\lim_{x\to{0}}-x^2 = \lim_{x\to{0}}x^2 = 0$, and thus, by squeeze theorem, we can conclude that $\lim_{x\to{0}}x^{n} = 0$ for all $n \ge 3$.
\end{solution}

	Let $L_1, \, L_2 \in \mathbb{R}$ such that $\lim_{x\to{c}} f = L_1$ and $\lim_{x\to{c}}(f + g) = L_2$. By algebraic limit theorems, we know that: $\lim_{x\to{c}}(f + g) - \lim_{x\to{c}}f = L_2 - L_1$ and thus:

\begin{subexercise}
	If $\lim_{x\to{c}}f$ and $\lim_{x\to{c}}f \cdot g$ exist, does it follow that $\lim_{x\to{c}}g$ exists?
\end{subexercise}

\begin{solution}
	No. Take for example $f(x) = x^2$ and $g(x) = \frac{1}{x}$. Noting that $f \cdot g = x^2 \cdot \frac{1}{x} = x$, we conclude that $\lim_{x\to{0}}f(x) = \lim_{x\to{0}} \left(f(x) \cdot g(x) \right) = 0$. However, $\lim_{x\to{c}}g = \lim_{x\to{c}} \frac{1}{x}$ does not have a limit in the real numbers. Therefore, it does not follow that $\lim_{x\to{c}}g$ exists even if $\lim_{x\to{c}}f$ and $\lim_{x\to{c}}\left(f \cdot g \right)$ do.
\end{solution}

\begin{exercise}[11]
	Determine whether the following limits exist in $\mathbb{R}$.
\end{exercise}

\begin{subexercise}
	$\lim_{x\to{0}} \sin \left( \frac{1}{x^2} \right) \; (x \neq 0)$.
\end{subexercise}

\begin{solution}
	This limit does not exist. Consider the two sequences $(x_n) = \frac{1}{\sqrt{2\pi{n} + \frac{\pi}{2}}}$ and $(y_n) = \frac{1}{\sqrt{2\pi{n} + \frac{3\pi}{2}}}$. We can see that trivially see that as $n$ approaches infinity, $(x_n)$ and $(y_n)$ approach 0. However, we can also observe that:
	\begin{align*}
		\lim_{n\to\infty}\sin \left( \frac{1}{(x_n)^2} \right) &= \lim_{n\to\infty} \sin\left( 2\pi{n} + \frac{\pi}{2} \right) \qquad &\lim_{n\to\infty} \sin \left( \frac{1}{(y_n)^2}\right) &= \sin \left( 2\pi{n} + \frac{3\pi}{2} \right) \\
		&= 1 \qquad &&= -1
	\end{align*}

	Thus, since there exists two sequences which both converge to zero but result in the function converging to two different limits, we know that $\lim_{n\to{0}}\sin \left( \frac{1}{x^2} \right)$ does not exist.
\end{solution}

\begin{subexercise}
	$\lim_{x\to{0}} x \sin \left( \frac{1}{x^2} \right) \; (x \neq 0)$.
\end{subexercise}

\begin{solution}
	This limit does exist. Since $-1 \le \sin \left( \frac{1}{x^2} \right) \le 1$, we ascertain that $-x \le \sin \left( \frac{1}{x^2} \right) \le x$. Since $\lim_{x\to{0}} -x = \lim_{x\to{0}} x = 0$, by the squeeze theorem, we know that $\lim_{x\to{0}}x\sin{\frac{1}{x^2}} = 0$.
\end{solution}

\begin{subexercise}
	$\lim_{x\to{0}}\sign \left( \sin \left( \frac{1}{x^2} \right) \right) \; (x \neq 0)$.
\end{subexercise}

\begin{solution}
	This limit does not exist. Once again, consider the two sequences $(x_n) = \frac{1}{\sqrt{2\pi{n} + \frac{\pi}{2}}}$ and $(y_n) = \frac{1}{\sqrt{2\pi{n} + \frac{3\pi}{2}}}$. Since both sequences converge to $0$ as $n$ goes to infinity, to show that $\lim_{x\to{0}} \sign \left( \sin \left( \frac{1}{x^2} \right) \right) \; (x \neq 0)$ does not exist, we can show that the function converges to two different values for each sequence. We know that $\sin \left( 2\pi{n} + \frac{\pi}{2} \right) = 1$ for all $n \in \mathbb{N}$ and similarly, $\sin \left( 2\pi{n} + \frac{3\pi}{2} \right) = -1$ for all $n \in \mathbb{N}$. Therefore:

	\begin{align*}
		\lim_{n\to\infty} \sign \left( \sin \left( \frac{1}{(x_n)^2} \right) \right) &= \lim_{n\to\infty} \sign \left( \sin \left( 2\pi{n} + \frac{\pi}{2} \right) \right) \\
		&= \lim_{n\to\infty} \sign \left( 1 \right) \\
		&= 1 \\\\
		\lim_{n\to\infty} \sign \left( \sin \left( \frac{1}{(y_n)^2} \right) \right) &= \lim_{n\to\infty} \sign \left( \sin \left( 2\pi{n} + \frac{3\pi}{2} \right) \right) \\
		&= \lim_{n\to\infty} \sign \left( -1 \right) \\
		&= -1
	\end{align*}

	Thus, since there exists two sequences which both converge to zero but result in the function converging to two different limits, we know that $\lim_{n\to{0}}\sign \left( \sin \left( \frac{1}{x^2} \right) \right)$ does not exist.
\end{solution}

Since $\lim_{x\to{0}^+} -\sqrt{x} = \lim_{x\to{0}^+} \sqrt{x} = 0$, by the squeeze theorem, we know that $\lim_{x\to{0}^+}\sqrt{x}\sin{\frac{1}{x^2}} = 0$.
\end{solution}

\begin{exercise}[15]
	Let $A \subseteq \mathbb{R}$ and let $f: A \rightarrow \mathbb{R}$, and let $c \in \mathbb{R}$ be a cluster point of $A$. In addition, suppose that $f(x) \ge 0$ for all $x \in A$, and let $\sqrt{f}$ be the function defined for $x \in A$ by $\left( \sqrt{f} \right)(x) \coloneq \sqrt{f(x)}$. If $\lim_{x\to{c}}$ exists, prove that $\lim_{x\to{c}} \sqrt{f} = \sqrt{\lim_{x\to{c}}f}$.
\end{exercise}

\begin{solution}
	\begin{proof}
		%TODO: Fill out wording
		Let $L = \lim_{x\to{c}} f$. By algebraic limit rules, we know that $\sqrt{L} = \lim_{x\to{c}} \sqrt{\lim_{x\to{c}} f}$. Since $\lim_{x\to{c}}f$ exists, then for arbitrary $\epsilon > 0$, there exists some $\delta > 0$ such that all $x \in A$ satisfying $0 < |x - c| < \delta$ also satisfy $|f(x) - f(c)| < \sqrt{L}\epsilon$. Therefore, for all $x$ satisfying $0 < |x - c| < \delta$, we have that:

		\begin{align*}
			|\sqrt{f} - \sqrt{L}| &= |\sqrt{f} - \sqrt{L}| \cdot \frac{\sqrt{f} + \sqrt{L}}{\sqrt{f} + \sqrt{L}} \\
			&= \frac{|\left(\sqrt{f} \right)^2 - L|}{\sqrt{f} + \sqrt{L}} \\
			&= \frac{|f - L|}{\sqrt{f} + \sqrt{L}} \\
			&\le \frac{|f - L|}{\sqrt{L}} \\
			&< \frac{\sqrt{L}\epsilon}{\sqrt{L}} \\
			&= \epsilon
		\end{align*}

		Therefore, since for arbitrary $\epsilon > 0$, there exists some $\delta > 0$ for which all $x$ satisfying $0 < |x - c| < \delta$ also satisfies $|\sqrt{f} - \sqrt{L}| < \epsilon$, by definition of functional limits, $\lim_{x\to{c}}\sqrt{f} = \sqrt{L}$. Since $\sqrt{L} = \sqrt{\lim_{x\to{c}}f}$, we thus have that:

		\begin{gather*}
			\lim_{x\to{c}}\sqrt{f} = \sqrt{\lim_{x\to{c}}f}
		\end{gather*}
	\end{proof}
\end{solution}

\begin{exercise}[4]
	Let $c \in \mathbb{R}$ and let $f$ be defined for $x \in \left( c, \infty \right)$ and $f(x) > 0$ for all $x \in \left( c, \infty \right)$. Show that $\lim_{x\to{c}}f = \infty$ if and only if $\lim_{x\to{c}}\frac{1}{f} = 0$.
\end{exercise}

\begin{solution}
	\begin{step}
		Forward direction.
	\end{step}

	If $\lim_{x\to{c}}f = \infty$, then we know that for all $M > 0$, there exists some $\delta > 0$ such that all $x$ satisfying $0 < |x - c| < \delta$ also satisfies $f(x) > M$. Let $\epsilon = \frac{1}{M}$, we can therefore see that, $\forall x \in V_{\delta}(c)$:

	\begin{align*}
		f(x) &> M \implies \\
		f(x) &> \frac{1}{\epsilon} \\
	\end{align*}

	Since $f(x) > 0$ for all $x \in \left( c, \infty \right)$, we know that $f(x) = |f(x)|$. Therefore, we can establish that:

	\begin{align*}
		f(x) &> \frac{1}{\epsilon} \implies \\
		|f(x)| &> \frac{1}{\epsilon} \implies \\
		\frac{1}{|f(x)|} &< \epsilon \\
		\left| \frac{1}{f(x)} \right| &< \epsilon \\
		\left| \frac{1}{f(x)} - 0 \right| &< \epsilon \\
	\end{align*}

	Therefore, since for all $\epsilon > 0$, there exists some $\delta > 0$ such that all $x$ satisfying $0 < |x - c| < \delta$ also satisfies $|\frac{1}{f(x)} - 0| < \epsilon$, by definition of functional limits, we know $\lim_{x\to{c}} \frac{1}{f} = 0$.

	\begin{step}
		Reverse Direction.
	\end{step}

	If $\lim_{x\to{c}}\frac{1}{f} = 0$, then we know that for all $\epsilon > 0$, there exists some $\delta > 0$ such that all $x$ satisfying $0 < |x - c| < \delta$ also satisfies $|\frac{1}{f} - 0| < \epsilon$. Therefore, we have for all such $x$ that:

	\begin{align*}
		|\frac{1}{f} - 0| &< \epsilon \implies \\
		|\frac{1}{f}| &< \epsilon \implies \\
		\frac{1}{|f|} &< \epsilon \implies \\
		|f| &> \epsilon \implies \\
	\end{align*}

	And since $f > 0$ for all $x$, we have that $f > \epsilon$. Thus, since for all $\epsilon > 0$, there exists some $\delta > 0$ for which $|f(x)| > \epsilon$ for all $0 < |x - c| < \delta$, we can conclude that $\lim_{x\to{c}}f = \infty$.
\end{solution}
